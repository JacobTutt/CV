% Compact, second-level details used beneath the relevant degree or role.
\newenvironment{cvdetails}[1]{%
  \vspace{-6mm}
  \raggedright
  \descriptionstyle{\textbf{#1}}\par
  \vspace{-2.4mm}
  \begin{list}{}{%
    \setlength{\leftmargin}{0pt}
    \setlength{\topsep}{0pt}
    \setlength{\itemsep}{0pt}
    \setlength{\parsep}{0pt}
    \setlength{\partopsep}{0pt}
  }
}{%
  \end{list}
  \vspace{2mm}
}

\newcommand*{\cvdetail}[2]{%
  \item[] \subdescriptionstyle{\textemdash\enskip}%
  \subentrytitlestyle{\textbf{#1}}%
  \ifempty{#2}{}{\subdescriptionstyle{\enskip---\enskip#2}}\par
}

}

% Compact publication / talk entry: title on one line, details beneath
\newsavebox{\cvpubtitlebox}
\newcommand*{\cvpub}[3]{%
  \item[] \sbox{\cvpubtitlebox}{\fontsize{9pt}{1em}\bodyfont\bfseries\color{darktext}#1}%
  \ifdim\wd\cvpubtitlebox>\linewidth\resizebox{\linewidth}{!}{\usebox{\cvpubtitlebox}}\else\usebox{\cvpubtitlebox}\fi\par
  \subdescriptionstyle{#2\ifempty{#3}{}{\enskip\textbullet\enskip\textit{#3}}}\par
  \vspace{1.0mm}
}


% Software entry: bold name + italic role on one line, italic description + link icon beneath
\newcommand*{\cvsoftware}[4]{%
  \item[] {\fontsize{11pt}{1em}\bodyfont\bfseries\upshape\color{darktext}\underline{#1}}\hfill{\fontsize{9pt}{1em}\bodyfont\itshape\color{graytext}#2}\par
  \vspace{0.3mm}
  {\fontsize{9pt}{1em}\bodyfont\itshape\color{text}#3}\hfill{\fontsize{9pt}{1em}\color{awesome}#4}\par
  \vspace{1mm}
}


% Talk entry: italic title, linked event, optional resource link, date flush right
\newcommand*{\cvlink}[2]{\ifempty{#1}{#2}{\href{#1}{\color{awesome}#2}}}
\newsavebox{\cvtalkbox}
\newcommand*{\cvtalk}[6]{%
  \item[] {\fontsize{9pt}{1.1em}\bodyfont\color{text}%
  \sbox{\cvtalkbox}{\textit{#1}}%
  \ifdim\wd\cvtalkbox>\dimexpr\linewidth-1.9cm\relax\resizebox{\dimexpr\linewidth-1.9cm\relax}{!}{\usebox{\cvtalkbox}}\else\usebox{\cvtalkbox}\fi
  \hfill{\itshape #6}}\par
  {\fontsize{9pt}{1.1em}\bodyfont\color{text}\cvlink{#4}{#3}, #5\ifempty{#2}{}{\enskip\textbullet\enskip#2}}\par
  \vspace{0.45mm}
}
% Teaching entry: module, programme and role, date flush right
\newcommand*{\cvteach}[4]{%
  \item[] {\fontsize{9pt}{1.1em}\bodyfont\color{text}%
  \parbox[t]{\dimexpr\linewidth-3.4cm\relax}{\raggedright\textit{#1}, #2, #3}\hfill{\itshape #4}}\par
  \vspace{0.4mm}
}
